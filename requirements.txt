# Runtime dependencies for Raspilapse.
#
# On a Raspberry Pi you do not need this file. Everything here arrives with
#
#     sudo apt install -y python3-picamera2 python3-yaml ffmpeg
#
# because python3-picamera2 depends on python3-numpy and python3-pil. Prefer
# the apt packages: picamera2 cannot be installed from pip at all, and pip
# upgrading numpy underneath it breaks it with a binary-incompatibility error.
#
# This file is for CI and for non-Pi development checkouts.

# Configuration. The one dependency with no other way in.
PyYAML>=6.0

# Overlay rendering and keograms; also the lores brightness metering the
# exposure loop closes on. Both are pulled in by python3-picamera2.
# Pillow's own advisories run past this, and the fix for the most recent of
# them is 12.2.0 -- which requires Python 3.10. This project supports 3.9,
# because Raspberry Pi OS Bullseye ships it and CI builds against it, so 11.x
# is the ceiling until 3.9 is dropped.
#
# In practice a Pi never installs this: apt's python3-pil is 9.4.0 and comes in
# with picamera2. This floor is for pip installs -- CI and development
# checkouts -- where nothing is pinned to the system libraries.
Pillow>=11.0.0
numpy>=1.21.0

# --- everything below is optional; the feature degrades if it is absent ---

# Sun elevation, recorded alongside each frame for the database and the
# solar graph script. Nothing decides from it.
# apt ships astral 1.6, whose API predates LocationInfo, so this one really
# does need pip: pip3 install --break-system-packages 'astral>=3.2'
astral>=3.2

# Uploading the daily video. requests-toolbelt streams the multipart body
# instead of holding ~300 MB in memory.
requests>=2.28.0
requests-toolbelt>=1.0.0

# scripts/db_graphs.py and scripts/graph_solar_patterns.py only.
matplotlib>=3.5.0
